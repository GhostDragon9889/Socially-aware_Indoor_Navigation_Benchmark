import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import time

root = Path.cwd()
out = Path(__file__).resolve().parent
source_paths = list((root / 'src').rglob('*.py')) + [root / 'scripts/run_external_python.py',
                out / 'run_robot_depth_observation.py', out / 'crowdsim_robot4_depth.yaml', Path(__file__).resolve()]
before = {str(p.relative_to(root)): hashlib.sha256(p.read_bytes()).hexdigest() for p in source_paths}
command = [sys.executable, '-m', 'coverage', 'run', '--rcfile=' + str(out.parent / 'coverage.rc'),
           'scripts/run_external_python.py', '--sdk-child', str(out / 'run_robot_depth_observation.py'),
           '--env-config', str(out / 'crowdsim_robot4_depth.yaml'), '--headless', '--scene-physics',
           '--total-steps', '80', '--rollout-steps', '4', '--ppo-epochs', '1', '--minibatch-size', '16',
           '--resume', '/project/EXP/SangforAndSaut/Socially-aware_Indoor_Navigation_Benchmark/output/external_sdk_continuation_20261011/robot_depth_serial/robot4_depth_train/20261012_010436/robot_ppo_latest.pt', '--save-interval', '16', '--no-wandb', '--output-dir', str(out / 'robot4_depth_train')]
record = {'command': command, 'status': 'running', 'actual_returncode': None, 'sources_before': before}
receipt = out / 'robot_depth_owner.json'
receipt.write_text(json.dumps(record, indent=2) + '\n')
started = time.monotonic()
with (out / 'robot4_depth_train.log').open('x') as log:
    process = subprocess.Popen(command, cwd=root, env=dict(os.environ,
        COVERAGE_FILE=str(out / '.coverage_robot_depth'), PYTHONDONTWRITEBYTECODE='1'),
        stdout=log, stderr=subprocess.STDOUT)
    code = process.wait()
record.update(actual_returncode=code, actual_returncode_source='subprocess.Popen.wait',
              duration_seconds=time.monotonic()-started,
              sources_unchanged=all(hashlib.sha256((root / p).read_bytes()).hexdigest()==digest for p,digest in before.items()),
              status='passed' if code == 0 else 'failed_preserved')
receipt.write_text(json.dumps(record, indent=2) + '\n')
raise SystemExit(code)
