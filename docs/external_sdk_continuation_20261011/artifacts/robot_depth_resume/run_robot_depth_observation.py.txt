"""Observe the unmodified production train loop; never substitute sensor data."""
import hashlib
import json
from pathlib import Path
import sys

from CrowdSim.ppo import train_ppo as training
import torch

OUT = Path(__file__).resolve().parent
record = {'schema': 'social_navigation.native_depth_training.v1',
          'python': sys.executable, 'status': 'running', 'observations': [],
          'scope': 'Actual Warehouse/NovaCarter, four robots, provided SMPL policy; short training, no convergence claim.'}
original = training.train_loop


def observed_loop(**arguments):
    nav, ppo = arguments['nav_manager'], arguments['ppo']
    assert nav.config.num_robots == 4 and nav.config.rl_depth_enabled
    observe = nav.get_robot_rl_observations
    before = {name: tensor.clone() for name, tensor in ppo.model.state_dict().items()}

    def observed_inputs():
        result = observe()
        raw = nav.env.crowdsim_robot_camera.data.output['distance_to_image_plane']
        depth = result[3]
        assert depth.shape == (4, 224, 224) and torch.isfinite(depth).all()
        assert 0 <= depth.min() <= depth.max() <= 1
        row = {'raw_shape': list(raw.shape), 'depth_shape': list(depth.shape),
               'raw_finite_nonzero_pixels': int((torch.isfinite(raw) & (raw > 0)).sum()),
               'depth_nonzero_pixels_per_robot': (depth > 0).sum((1, 2)).tolist(),
               'depth_max': float(depth.max())}
        assert min(row['depth_nonzero_pixels_per_robot']) > 0, 'Real depth required for every robot'
        record['observations'].append(row)
        (OUT / 'robot_depth_observation.json').write_text(json.dumps(record, indent=2) + '\n')
        return result

    nav.get_robot_rl_observations = observed_inputs
    try:
        original(**arguments)
        flags = [name for name in ('_warned_depth_fallback', '_warned_depth_unavailable',
                                  '_warned_depth_read_error', '_warned_depth_dim') if getattr(nav, name, False)]
        assert not flags, 'Fallback depth is not verification: ' + repr(flags)
        groups = {}
        for prefix in ('vector_encoder.', 'depth_encoder.', 'map_encoder.', 'fusion.', 'critic.'):
            groups[prefix] = [name for name, value in ppo.model.state_dict().items()
                              if name.startswith(prefix) and not torch.equal(value, before[name])]
            assert groups[prefix], 'No learned parameter changed: ' + prefix
        checkpoint = arguments['output_dir'] / 'robot_ppo_latest.pt'
        with torch.serialization.safe_globals([training.RobotPPOConfig]):
            value = torch.load(checkpoint, map_location='cpu', weights_only=True)
        record.update(status='passed', changed_parameter_groups=groups,
                      output=str(arguments['output_dir']), checkpoint_sha256=hashlib.sha256(checkpoint.read_bytes()).hexdigest(),
                      checkpoint_keys=list(value), depth_fallback_flags=flags)
    except BaseException as error:
        record.update(status='failed_preserved', error=type(error).__name__ + ': ' + str(error))
        raise
    finally:
        (OUT / 'robot_depth_observation.json').write_text(json.dumps(record, indent=2) + '\n')


training.train_loop = observed_loop
training.main()
